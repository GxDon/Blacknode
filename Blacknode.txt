#!/usr/bin/env python3

import os
from flask import Flask, render_template


app = Flask(__name__)

LOG_FILE = "logs/monitor.log"

os.makedirs("logs", exist_ok=True)


def leer_logs():
    if not os.path.isfile(LOG_FILE):
        return []

    try:
        with open(
            LOG_FILE,
            "r",
            encoding="utf-8",
            errors="replace"
        ) as file:
            lineas = file.readlines()

        return lineas[-20:][::-1]

    except OSError as error:
        print(f"[!] Error leyendo logs: {error}")
        return []


@app.route("/")
def index():
    logs = leer_logs()

    return render_template(
        "index.html",
        logs=logs
    )


@app.route("/health")
def health():
    return {
        "status": "ok",
        "service": "Blacknode"
    }


if __name__ == "__main__":

    print("=" * 50)
    print("              Blacknode")
    print("=" * 50)

    app.run(
        host="127.0.0.1",
        port=5000,
        debug=False
    )
